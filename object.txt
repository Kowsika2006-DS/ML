import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from tensorflow.keras.datasets import cifar10
from tensorflow.keras.utils import to_categorical
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Dropout, Conv2D, MaxPool2D, Flatten

# Load CIFAR-10 dataset
(X_train, y_train), (X_test, y_test) = cifar10.load_data()
classes = ['airplane', 'automobile', 'bird', 'cat', 'deer',
           'dog', 'frog', 'horse', 'ship', 'truck']

# Show 9 training images
plt.figure(figsize=(10, 10))
for i in range(9):
    plt.subplot(3, 3, i + 1)
    plt.title(f'{y_train[i, 0]}-{classes[y_train[i, 0]]}')
    plt.imshow(X_train[i])
plt.suptitle('Examples of training images and their labels')
plt.show()

# Preprocess data
y_train = to_categorical(y_train, 10)
y_test = to_categorical(y_test, 10)
X_train = X_train.astype('float32') / 255
X_test = X_test.astype('float32') / 255

print('Training:', X_train.shape, y_train.shape)
print('Testing:', X_test.shape, y_test.shape)

# Build CNN model
model = Sequential([
    Conv2D(25, (3, 3), padding='same', activation='relu',
           input_shape=(32, 32, 3)),
    Conv2D(50, (3, 3), padding='same', activation='relu'),
    MaxPool2D((2, 2)),
    Dropout(0.25),
    Conv2D(70, (3, 3), padding='same', activation='relu'),
    MaxPool2D((2, 2)),
    Dropout(0.25),
    Flatten(),
    Dense(500, activation='relu'),
    Dropout(0.4),
    Dense(250, activation='relu'),
    Dropout(0.3),
    Dense(10, activation='softmax')
])

model.compile(loss='categorical_crossentropy',
              optimizer='adam', metrics=['accuracy'])
model.summary()

# Train the model (same 50 epochs)
history = model.fit(
    X_train, y_train, batch_size=40, epochs=50,
    validation_data=(X_test, y_test)
)

# Plot accuracy and loss curves
fig, ax = plt.subplots(1, 2, figsize=(15, 5))
ax[0].plot(history.history['accuracy'], label='Training')
ax[0].plot(history.history['val_accuracy'], label='Validation')
ax[0].set_title('Accuracy curves')
ax[0].set_xlabel('Epochs')
ax[0].set_ylabel('Accuracy')
ax[0].grid()
ax[0].legend()

ax[1].plot(history.history['loss'], label='Training')
ax[1].plot(history.history['val_loss'], label='Validation')
ax[1].set_title('Loss curves')
ax[1].set_xlabel('Epochs')
ax[1].set_ylabel('Loss')
ax[1].grid()
ax[1].legend()
plt.tight_layout()
plt.show()

# Predict and display 15 testing images
m = 15
y_pred = model.predict(X_test[:m], verbose=0).argmax(axis=1)
plt.figure(figsize=(12, 6))
plt.suptitle('Testing images and their labels')
for i in range(m):
    plt.subplot(3, 5, i + 1)
    plt.title(classes[y_pred[i]])
    plt.imshow(X_test[i])
    plt.axis('off')
plt.tight_layout()
plt.show()

# Count predicted classes
unique, counts = np.unique(y_pred, return_counts=True)
print(pd.DataFrame({
    'class name': [classes[i] for i in unique],
    'count': counts
}))
